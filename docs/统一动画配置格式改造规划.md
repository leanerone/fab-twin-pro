# 统一动画配置格式改造规划（平台侧）

> 版本：v1.0
> 日期：2026-08-18
> 适用对象：平台开发
> 目标：废弃当前 `animation_config_json` 旧格式，全量切换为通用 Motion JSON v1.1 格式

---

## 一、改造目标

| 项目 | 当前状态 | 目标状态 |
|------|---------|---------|
| 配置格式 | `{ machine_type, flows, animations, targets }` | `{ schema_version, document, parts, motions, ext }` |
| 事件触发 | `event_to_phase` 查表（1事件→1动画） | `motions[].step + rules[].when` 条件匹配（1事件→多分支） |
| 坐标系统 | 无声明，散落各组件硬编码 | `document.coord` 显式声明，Y轴向上 |
| 动作类型 | translate/rotate/scale/flash/visibility/color/scan/signal | offset/rotate/scale/opacity/visibility/flash/color/scan/signal |
| 存储方式 | `animation_config_json` 字段存旧格式 | 同字段存通用 Motion JSON |
| 编辑界面 | Tab2 编辑 flows/phases/event_to_phase/animations/targets | Tab2 编辑 parts/motions/rules/actions |

## 三、分阶段实施计划

### Phase 1：基础设施（新增3个 composable）

**目标**：建立通用 Motion JSON 的解析、求值、坐标转换能力，不影响现有功能。

#### 1.1 新增 `useExprEval.js`（when 表达式求值器）

**路径**：`frontend/src/composables/useExprEval.js`

**职责**：解析 `when` 条件表达式，返回 boolean。

```javascript
// 输入：when 字符串 + event payload
// 输出：boolean
// 支持运算符：== != > < >= <= && || !
// 支持：字符串(单引号) 数字 true false
// params 字段自动从 event payload 提取（下划线转驼峰）

export function evalWhen(when, eventPayload) {
  if (when === 'true' || !when) return true
  const params = extractParams(eventPayload)
  const expr = when.replace(/params\.(\w+)/g, (_, key) => {
    const val = params[key]
    return typeof val === 'string' ? `'${val}'` : String(val)
  })
  // 安全求值：仅允许比较和逻辑运算
  return safeEval(expr)
}

function extractParams(payload) {
  // chamber_id → chamber, port_id → port, wafer_id → wafer
  const params = {}
  for (const [k, v] of Object.entries(payload || {})) {
    const camelKey = k.replace(/_id$/, '').replace(/_(\w)/g, (_, c) => c.toUpperCase())
    params[camelKey] = v
  }
  return params
}
```

**验证方式**：
```javascript
evalWhen("params.chamber == '1'", { chamber_id: '1' })  // true
evalWhen("params.chamber == '2'", { chamber_id: '1' })  // false
evalWhen("true", {})  // true
```

#### 1.2 新增 `useCoordTransform.js`（坐标转换层）

**路径**：`frontend/src/composables/useCoordTransform.js`

**职责**：逻辑坐标系（Y向上，原点左下）↔ SVG 浏览器坐标系（Y向下，原点左上）互转。

```javascript
export function logicalToSvg(x, y, viewBoxH) {
  return { svgX: x, svgY: viewBoxH - y }
}

export function svgToLogical(x, y, viewBoxH) {
  return { logicalX: x, logicalY: viewBoxH - y }
}

// rotate pivot 转换
export function pivotLogicalToSvg(pivot, viewBoxH) {
  return { x: pivot.x, y: viewBoxH - pivot.y }
}

// offset 转换：Y 方向取反
export function offsetLogicalToSvg(offset) {
  return { offsetX: offset.offset_x || 0, offsetY: -(offset.offset_y || 0) }
}
```

#### 1.3 新增 `useMotionConfig.js`（配置加载器，替代 useAnimationConfig）

**路径**：`frontend/src/composables/useMotionConfig.js`

**职责**：从 DB 加载通用 Motion JSON，提供查询 API。

```javascript
import { ref } from 'vue'
import { api } from '../api'
import { evalWhen } from './useExprEval'

const configCache = new Map()

export function useMotionConfig(modelId) {
  const config = ref(null)
  const loading = ref(false)

  async function load() {
    if (configCache.has(modelId)) {
      config.value = configCache.get(modelId)
      return
    }
    loading.value = true
    const model = await api.getModel(modelId)
    const raw = model.animation_config_json
    if (raw && raw.schema_version) {
      config.value = raw
      configCache.set(modelId, raw)
    } else if (raw && raw.machine_type) {
      // 旧格式，触发迁移或回退
      config.value = migrateOldFormat(raw)
      configCache.set(modelId, config.value)
    }
    loading.value = false
  }

  // 按事件名查找匹配的 motions
  function getMotionsByStep(stepName) {
    if (!config.value) return []
    return config.value.motions.filter(m => m.step === stepName && m.enabled !== false)
  }

  // 匹配 rule（返回命中 rule 的 actions）
  function matchActions(stepName, eventPayload) {
    const motions = getMotionsByStep(stepName)
    for (const motion of motions) {
      for (const rule of motion.rules || []) {
        if (evalWhen(rule.when, eventPayload)) {
          return {
            target_part_id: rule.target_part_id || motion.target_part_id,
            actions: rule.actions || [],
          }
        }
      }
    }
    return { target_part_id: null, actions: [] }
  }

  // 查找部件定义
  function getPart(partId) {
    if (!config.value) return null
    return config.value.parts.find(p => p.part_id === partId)
  }

  // 获取坐标声明
  function getCoord() {
    return config.value?.document?.coord || null
  }

  return { config, loading, load, getMotionsByStep, matchActions, getPart, getCoord }
}
```

### Phase 2：事件处理重构

**目标**：`useEventActionMapping.js` 改为消费 `motions/rules`，替代 `event_to_phase` 查表。

#### 2.1 改造 `useEventActionMapping.js`

**当前逻辑**：
```
event → parseEventAction() → 查 event_action_mappings 或 EVENT_ACTION_DEFAULTS → 返回 action
```

**改造后逻辑**：
```
event → useMotionConfig.matchActions(event_name, event.payload) → 返回 actions[]
        → 遍历 actions，按 type 执行（offset→translate, rotate→rotate, flash→flash...）
```

**关键改动**：

```javascript
// 旧：parseEventAction 查表返回单个 action
// 新：matchActions 返回多个 actions，逐个执行

export function useEventActionMapping(props) {
  const { matchActions, getPart, getCoord } = useMotionConfig(props.modelId)
  const currentActions = ref([])
  const chamberState = ref({})

  function processEvent(event) {
    const eventName = event.event_name || event.event_code
    const { target_part_id, actions } = matchActions(eventName, event.payload)

    for (const action of actions) {
      executeAction(target_part_id, action)
    }
  }

  function executeAction(partId, action) {
    switch (action.type) {
      case 'offset': applyOffset(partId, action); break
      case 'rotate': applyRotate(partId, action); break
      case 'scale': applyScale(partId, action); break
      case 'opacity': applyOpacity(partId, action); break
      case 'visibility': applyVisibility(partId, action); break
      case 'flash': applyFlash(partId, action); break
      case 'color': applyColor(partId, action); break
      case 'scan': applyScan(partId, action); break
      case 'signal': applySignal(partId, action); break
      case 'custom': applyCustom(partId, action); break
    }
  }

  return { currentActions, chamberState, processEvent }
}
```

#### 2.2 保留 `EVENT_ACTION_DEFAULTS` 作为兜底

当 `useMotionConfig` 未找到匹配 motion 时，回退到 `EVENT_ACTION_DEFAULTS` 硬编码映射，确保未配置新格式的机型仍能工作。

### Phase 3：视图组件适配

**目标**：MachineModel2D/VpoView/OxeView 消费新格式驱动动画。

#### 3.1 MachineModel2D.vue 改造

**当前**：硬编码 POS 坐标 + 硬编码角度映射。

**改造后**：
1. 从 `useMotionConfig.getCoord()` 获取 viewBox
2. 从 `useMotionConfig.getPart(partId)` 获取部件信息
3. 用 `useCoordTransform` 转换坐标
4. 用 `useEventActionMapping.processEvent(event)` 触发动画

```javascript
// 旧：const POS = { smif1: {x:120,y:430}, ... }  硬编码
// 新：从 SVG 元素实际位置读取，或从 parts.anchors 读取

import { useMotionConfig } from '../composables/useMotionConfig'
import { useCoordTransform } from '../composables/useCoordTransform'

const { getCoord, getPart } = useMotionConfig(modelId)

function applyOffset(partId, action) {
  const el = svgRef.value.querySelector(`#${partId}`)
  if (!el) return
  const coord = getCoord()
  const { offsetX, offsetY } = useCoordTransform.offsetLogicalToSvg(action)
  el.style.transform = `translate(${offsetX}px, ${offsetY}px)`
  el.style.transition = `transform ${action.duration || 0}ms ${action.easing || 'linear'}`
}

function applyRotate(partId, action) {
  const el = svgRef.value.querySelector(`#${partId}`)
  if (!el) return
  const coord = getCoord()
  const pivot = useCoordTransform.pivotLogicalToSvg(action.pivot, coord.viewBox[3])
  el.style.transformOrigin = `${pivot.x}px ${pivot.y}px`
  el.style.transform = `rotate(${action.angle}deg)`
  el.style.transition = `transform ${action.duration || 0}ms ${action.easing || 'linear'}`
}
```

### Phase 5：数据迁移

**目标**：将旧格式 podopener.json 转为通用 Motion JSON v1.1。

#### 5.1 迁移映射规则

```
旧格式                              → 通用格式
{
  machine_type: "PODOPENER",        → ext.machine_type: "PODOPENER"
  version: "1.0",                   → ext.version: "1.0"
  flows.PACKING.phases[]:           → motions[]（每个 phase 变成一个 motion）
    { key: "POD_PLACE",             →   step: "POD_PLACED"（用 event_to_phase 反查事件名）
      duration_ms: 800 }            →   rules: [{ when: "true", actions: [...] }]
  flows.PACKING.event_to_phase:     → 反向映射：event → step
    "POD_PLACED": { phase: "POD_PLACE",
                    anim: "pod.enter" }
  animations["pod.enter"]:          → rules[0].actions[0]
    { target: "podShell",           →   target_part_id: "podShell"
      action: "translate",          →   type: "offset"
      axis: "z",                    →   offset_z: (to - from)
      from: 148, to: 680,           →   （差值 = 532）
      easing: "mechanical" }        →   easing: "mechanical"
  targets["podShell"]:              → parts[]: { part_id: "podShell", ... }
    { view_2d: "podShell",          →   part_id: "podShell"
      view_3d: "podShell",          →   view_3d: "podShell"
      desc: "POD外壳" }              →   desc: "POD外壳"
  无 coord                          → document.coord: { type:"svg_user_space",
                                    →   origin:"bottom_left", y_axis:"up",
                                    →   viewBox: [0,0,700,500] }
```

#### 5.2 迁移脚本

**路径**：`scripts/migrate_to_motion_json.py`

```python
"""
将旧格式 animation_config_json 迁移为通用 Motion JSON v1.1
用法：
  python migrate_to_motion_json.py --model-id podopener --dry-run  # 预览
  python migrate_to_motion_json.py --model-id podopener             # 执行
"""
```

**迁移步骤**：
1. 从 DB 读取 `animation_config_json`
2. 检测 `schema_version` 字段，已有则跳过
3. 解析 flows → 反向构建 event→step 映射
4. 遍历 phases，每个 phase 生成一个 motion
5. 用 event_to_phase 反查该 phase 对应的事件名作为 step
6. 查找该 phase 关联的 animation，转为 actions
7. targets → parts 转换
8. 补充 document.coord（从已知 SVG viewBox）
9. 写回 `animation_config_json`

## 五、风险与对策

| 风险 | 影响 | 对策 |
|------|------|------|
| 旧机型未迁移导致动画失效 | PODOPENER 等机型回放异常 | Phase 2 保留 `EVENT_ACTION_DEFAULTS` 兜底，Phase 5 完成迁移后才进 Phase 6 |
| `when` 表达式注入攻击 | 恶意 payload 执行任意代码 | `useExprEval` 用白名单 AST 求值，禁止 eval/Function |
| 坐标系转换错误 | 部件位置偏移 | `useCoordTransform` 提供 `logicalToSvg`/`svgToLogical` 双向转换 + 单元测试 |
| Motion JSON 格式不合规 | 运行时解析失败 | ModelEditor 导入时校验 `schema_version` + 必填字段 + 提示错误位置 |
| DB 旧格式数据残留 | 混合格式导致逻辑混乱 | 迁移脚本检测 `schema_version`，无则标记为待迁移 |
| 模型组 step 名与平台事件名不一致 | 事件触发失败 | Phase 3 后增加校验：加载时比对 step 与已知事件名，不匹配则告警 |
